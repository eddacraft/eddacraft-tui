```typescript
export interface CommandRule {

  // Documentation
```

```typescript
// core/src/gate/parsers/command-parser.ts

import { parse as parseShellCommand } from 'shell-quote';
import type { ParsedCommand } from '../rules/types.js';

export class CommandParser {
  parse(cmd: string): ParsedCommand {
    // 1. Unwrap shell wrappers
    const { unwrapped, wrappers } = this.unwrap(cmd);

    // 2. Tokenise using shell-quote
    const tokens = parseShellCommand(unwrapped);
    const stringTokens = tokens.map((t) => String(t));

    // 3. Extract command, subcommand, flags, args
    const [command, ...rest] = stringTokens;
    const flags = rest.filter((t) => t.startsWith('-'));
    const args = rest.filter((t) => !t.startsWith('-'));
    const subcommand = args[0];

    return {
      raw: cmd,
      command,
      subcommand,
      flags,
      args,
      unwrapped,
      wrapperChain: wrappers,
    };
  }

  private unwrap(
    cmd: string,
    depth = 0
  ): { unwrapped: string; wrappers: string[] } {
    if (depth > 5) {
      return { unwrapped: cmd, wrappers: [] };
    }

    const trimmed = cmd.trim();

    // Shell wrappers
    const shellPatterns: Array<[RegExp, string]> = [
      [/^bash\s+-c\s+["'](.+)["']$/, 'bash'],
      [/^sh\s+-c\s+["'](.+)["']$/, 'sh'],
      [/^sudo\s+(.+)$/, 'sudo'],
      [/^env\s+(?:\w+=\w+\s+)*(.+)$/, 'env'],
      [/^command\s+(?:-[pvV]\s+)*(.+)$/, 'command'],
    ];

    for (const [pattern, wrapper] of shellPatterns) {
      const match = trimmed.match(pattern);
      if (match?.[1]) {
        const inner = this.unwrap(match[1], depth + 1);
        return {
          unwrapped: inner.unwrapped,
          wrappers: [wrapper, ...inner.wrappers],
        };
      }
    }

    // Interpreter one-liners
    const interpreterPatterns: Array<[RegExp, string]> = [
      [/python\s+-c\s+["'].*?system\(["'](.+?)["']\)/, 'python'],
      [/node\s+-e\s+["'].*?exec\(["'](.+?)["']\)/, 'node'],
    ];

    for (const [pattern, wrapper] of interpreterPatterns) {
      const match = trimmed.match(pattern);
      if (match?.[1]) {
        const inner = this.unwrap(match[1], depth + 1);
        return {
          unwrapped: inner.unwrapped,
          wrappers: [wrapper, ...inner.wrappers],
        };
      }
    }

    return { unwrapped: trimmed, wrappers: [] };
  }
}
```

### 3. Implement rule matcher (CMDSAF-003)

Create rule matcher that finds most-specific matching rule for a parsed command.

**Files to create:**

- `core/src/gate/rules/rule-matcher.ts`
- `core/src/gate/rules/rule-matcher.test.ts`

**Checkpoint:** Matcher correctly identifies matching rules with proper
specificity ordering

**Validate:**

```bash
nx test core --testNamePattern="RuleMatcher"
```

**Pattern:** Specificity: command (1) + subcommand (2) + flags (4) + args (8)

**Implementation:**

```typescript
// core/src/gate/rules/rule-matcher.ts

import type { CommandRule, ParsedCommand } from './types.js';

export class RuleMatcher {
  findMatchingRule(
    parsed: ParsedCommand,
    rules: CommandRule[]
  ): CommandRule | undefined {
    // Sort by specificity (most specific first)
    const sorted = this.sortBySpecificity(rules);

    // Find first matching rule
    for (const rule of sorted) {
      if (this.matches(parsed, rule)) {
        return rule;
      }
    }

    return undefined;
  }

  private matches(parsed: ParsedCommand, rule: CommandRule): boolean {
    // 1. Command match
    if (parsed.command !== rule.command) {
      return false;
    }

    // 2. Subcommand match (if specified)
    if (rule.subcommand && parsed.subcommand !== rule.subcommand) {
      return false;
    }

    // 3. Flag match
    if (rule.flags) {
      if (!this.matchFlags(parsed.flags, rule.flags)) {
        return false;
      }
    }

    // 4. Argument match
    if (rule.args) {
      if (!this.matchArgs(parsed.args, rule.args)) {
        return false;
      }
    }

    return true;
  }

  private matchFlags(
    parsedFlags: string[],
    ruleFlags: NonNullable<CommandRule['flags']>
  ): boolean {
    // Required flags: ALL must be present
    if (ruleFlags.required) {
      const hasAllRequired = ruleFlags.required.every((required) =>
        parsedFlags.some((flag) => this.flagMatches(flag, required))
      );
      if (!hasAllRequired) {
        return false;
      }
    }

    // Forbidden flags: NONE can be present
    if (ruleFlags.forbidden) {
      const hasForbidden = ruleFlags.forbidden.some((forbidden) =>
        parsedFlags.some((flag) => this.flagMatches(flag, forbidden))
      );
      if (hasForbidden) {
        return false;
      }
    }

    // Dangerous flags: ANY triggers match
    if (ruleFlags.dangerous) {
      const hasDangerous = ruleFlags.dangerous.some((dangerous) =>
        parsedFlags.some((flag) => this.flagMatches(flag, dangerous))
      );
      if (!hasDangerous) {
        return false;
      }
    }

    return true;
  }

  private flagMatches(flag: string, pattern: string): boolean {
    // Handle combined short flags: -rf contains -r and -f
    if (flag.startsWith('-') && !flag.startsWith('--')) {
      return flag.includes(pattern.replace('-', ''));
    }
    return flag === pattern || flag.includes(pattern);
  }

  private matchArgs(
    parsedArgs: string[],
    ruleArgs: NonNullable<CommandRule['args']>
  ): boolean {
    if (ruleArgs.pattern) {
      // Match pattern against specific position or any arg
      if (ruleArgs.position !== undefined) {
        const arg = parsedArgs[ruleArgs.position];
        return arg ? ruleArgs.pattern.test(arg) : false;
      } else {
        return parsedArgs.some((arg) => ruleArgs.pattern!.test(arg));
      }
    }
    return true;
  }

  private sortBySpecificity(rules: CommandRule[]): CommandRule[] {
    return [...rules].sort((a, b) => {
      const scoreA = this.calculateSpecificity(a);
      const scoreB = this.calculateSpecificity(b);
      return scoreB - scoreA; // Most specific first
    });
  }

  private calculateSpecificity(rule: CommandRule): number {
    let score = 0;
    score += 1; // Base: has command
    if (rule.subcommand) score += 2;
    if (rule.flags) score += 4;
    if (rule.args) score += 8;
    return score;
  }
}
```

```typescript
describe('RuleMatcher', () => {
  const matcher = new RuleMatcher();

  describe('basic matching', () => {
    it('matches on command only', () => {
      const rules: CommandRule[] = [
        {
          id: 'git-any',
          category: 'git',
          command: 'git',
          action: 'allow',
          severity: 'info',
          reason: 'Git is safe',

      ];

      const parsed: ParsedCommand = {

        subcommand: 'status',
        flags: [],
        args: ['status'],

        raw: 'git push --force-with-lease',
        command: 'git',
        subcommand: 'push',
        flags: ['--force-with-lease'],
        args: ['push'],
        unwrapped: 'git push --force-with-lease',
        wrapperChain: [],
      };
      expect(matcher.findMatchingRule(parsed2, rules)).toBeUndefined();
    });
  });

  describe('specificity ordering', () => {
    it('matches most specific rule first', () => {
      const rules: CommandRule[] = [
        {
          id: 'git-any',
          category: 'git',
          command: 'git',

        args: ['reset'],
        unwrapped: 'git reset --hard',
        wrapperChain: [],
      };

      // Should match the more specific rule
      const match = matcher.findMatchingRule(parsed, rules);
      expect(match?.id).toBe('git-reset-hard');
    });
  });
```
