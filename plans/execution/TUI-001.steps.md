### 4. Implement TTY Detection Utility

**File:** `ui/src/utils/tty-detection.ts`

```typescript
/**
 * Detect whether TUI should be enabled based on environment
 */
export interface TUIDetectionOptions {
  tui?: boolean;
  noTui?: boolean;
  json?: boolean;
  quiet?: boolean;
  ci?: boolean;
}

export function shouldUseTUI(options: TUIDetectionOptions = {}): boolean {
  // Explicit flag takes precedence
  if (options.tui !== undefined) return options.tui;
  if (options.noTui) return false;

  // Incompatible options
  if (options.json || options.quiet) return false;

  // Auto-detect TTY and environment
  return process.stdout.isTTY === true && !process.env.CI && !options.ci;
}

export interface TerminalSize {
  columns: number;
  rows: number;
}

export function getTerminalSize(): TerminalSize {
  return {
    columns: process.stdout.columns ?? 80,
    rows: process.stdout.rows ?? 24,
  };
}

export function isTerminalSizeAdequate(minColumns = 80, minRows = 20): boolean {
  const size = getTerminalSize();
  return size.columns >= minColumns && size.rows >= minRows;
}
```

```typescript
import React from 'react';
import { Text } from 'ink';

export interface StatusBadgeProps {
  status: 'passed' | 'failed' | 'running' | 'skipped';
  label?: string;
}

export const StatusBadge: React.FC<StatusBadgeProps> = ({ status, label }) => {
  const symbols = {
    passed: '✓',
    failed: '✗',
    running: '⣾',
    skipped: '⊘',
  };

  const colours = {
    passed: 'green',
    failed: 'red',
    running: 'blue',
    skipped: 'yellow',
  };

  return (
    <Text color={colours[status]}>
      {symbols[status]} {label ?? status.toUpperCase()}
    </Text>
  );
};
```

**File:** `ui/src/components/ProgressBar.tsx`

```typescript
import React from 'react';
import { Box, Text } from 'ink';

export interface ProgressBarProps {
  progress: number; // 0-100
  label?: string;
  width?: number;
  showPercentage?: boolean;
}

export const ProgressBar: React.FC<ProgressBarProps> = ({
  progress,
  label,
  width = 40,
  showPercentage = true,
}) => {
  const clampedProgress = Math.max(0, Math.min(100, progress));
  const filledWidth = Math.round((width * clampedProgress) / 100);
  const emptyWidth = width - filledWidth;

  const filled = '█'.repeat(filledWidth);
  const empty = '░'.repeat(emptyWidth);

  return (
    <Box flexDirection="row" gap={1}>
      {label && <Text>{label}</Text>}
      <Text color="cyan">
        {filled}
        {empty}
      </Text>
      {showPercentage && <Text>{clampedProgress.toFixed(0)}%</Text>}
    </Box>
  );
};
```

### 7. Write Unit Tests

**File:** `ui/src/utils/__tests__/tty-detection.test.ts`

```typescript
import { describe, it, expect, beforeEach, afterEach } from 'vitest';
import {
  shouldUseTUI,
  getTerminalSize,
  isTerminalSizeAdequate,
} from '../tty-detection.js';

describe('TTY Detection', () => {
  let originalIsTTY: boolean | undefined;
  let originalCI: string | undefined;

  beforeEach(() => {
    originalIsTTY = process.stdout.isTTY;
    originalCI = process.env.CI;
  });

  afterEach(() => {
    if (originalIsTTY !== undefined) {
      process.stdout.isTTY = originalIsTTY;
    } else {
      delete (process.stdout as any).isTTY;
    }
    if (originalCI !== undefined) {
      process.env.CI = originalCI;
    } else {
      delete process.env.CI;
    }
  });

  describe('shouldUseTUI', () => {
    it('returns true when TTY and not CI', () => {
      process.stdout.isTTY = true;
      delete process.env.CI;
      expect(shouldUseTUI()).toBe(true);
    });

    it('returns false when --no-tui flag set', () => {
      process.stdout.isTTY = true;
      expect(shouldUseTUI({ noTui: true })).toBe(false);
    });

    it('returns true when --tui flag explicitly set', () => {
      process.stdout.isTTY = false;
      expect(shouldUseTUI({ tui: true })).toBe(true);
    });

    it('returns false in CI environment', () => {
      process.stdout.isTTY = true;
      process.env.CI = 'true';
      expect(shouldUseTUI()).toBe(false);
    });

    it('returns false when --json flag set', () => {
      process.stdout.isTTY = true;
      expect(shouldUseTUI({ json: true })).toBe(false);
    });

    it('returns false when --quiet flag set', () => {
      process.stdout.isTTY = true;
      expect(shouldUseTUI({ quiet: true })).toBe(false);
    });
  });

  describe('getTerminalSize', () => {
    it('returns current terminal size', () => {
      const size = getTerminalSize();
      expect(size).toHaveProperty('columns');
      expect(size).toHaveProperty('rows');
      expect(typeof size.columns).toBe('number');
      expect(typeof size.rows).toBe('number');
    });

    it('returns default values when stdout has no size', () => {
      const originalColumns = process.stdout.columns;
      const originalRows = process.stdout.rows;
      delete (process.stdout as any).columns;
      delete (process.stdout as any).rows;

      const size = getTerminalSize();
      expect(size.columns).toBe(80);
      expect(size.rows).toBe(24);

      process.stdout.columns = originalColumns;
      process.stdout.rows = originalRows;
    });
  });

  describe('isTerminalSizeAdequate', () => {
    it('returns true for adequate size', () => {
      const originalColumns = process.stdout.columns;
      const originalRows = process.stdout.rows;
      process.stdout.columns = 120;
      process.stdout.rows = 30;

      expect(isTerminalSizeAdequate()).toBe(true);

      process.stdout.columns = originalColumns;
      process.stdout.rows = originalRows;
    });

    it('returns false for inadequate size', () => {
      const originalColumns = process.stdout.columns;
      const originalRows = process.stdout.rows;
      process.stdout.columns = 60;
      process.stdout.rows = 15;

      expect(isTerminalSizeAdequate()).toBe(false);

      process.stdout.columns = originalColumns;
      process.stdout.rows = originalRows;
    });
  });
});
```
