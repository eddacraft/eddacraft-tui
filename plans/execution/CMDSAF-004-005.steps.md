// core/src/gate/rules/default-git-rules.test.ts

import { describe, it, expect } from 'vitest';
import { DEFAULT_GIT_RULES } from './default-git-rules.js';
import { CommandParser } from '../parsers/command-parser.js';
import { RuleMatcher } from './rule-matcher.js';

describe('Default Git Rules', () => {
  const parser = new CommandParser();
  const matcher = new RuleMatcher();

  function analyseCommand(cmd: string) {
    const parsed = parser.parse(cmd);
    const rule = matcher.findMatchingRule(parsed, DEFAULT_GIT_RULES);
    return { parsed, rule };
  }

  describe('destructive operations - BLOCKED', () => {
    it('blocks git reset --hard', () => {
      const { rule } = analyseCommand('git reset --hard');
      expect(rule?.action).toBe('block');
      expect(rule?.reason).toContain('destroys uncommitted changes');
    });

    it('blocks git reset --hard HEAD~1', () => {
      const { rule } = analyseCommand('git reset --hard HEAD~1');
      expect(rule?.action).toBe('block');
    });

    it('blocks git checkout -- file.txt', () => {
      const { rule } = analyseCommand('git checkout -- file.txt');
      expect(rule?.action).toBe('block');
      expect(rule?.reason).toContain('discards uncommitted changes');
    });

    it('blocks git checkout -- .', () => {
      const { rule } = analyseCommand('git checkout -- .');
      expect(rule?.action).toBe('block');
    });

    it('blocks git restore file.txt', () => {
      const { rule } = analyseCommand('git restore file.txt');
      expect(rule?.action).toBe('block');
    });

    it('blocks git restore --worktree file.txt', () => {
      const { rule } = analyseCommand('git restore --worktree file.txt');
      expect(rule?.action).toBe('block');
    });

    it('blocks git push --force', () => {
      const { rule } = analyseCommand('git push --force origin main');
      expect(rule?.action).toBe('block');
      expect(rule?.reason).toContain('rewrites remote history');
    });

    it('blocks git push -f (short form)', () => {
      const { rule } = analyseCommand('git push -f');
      expect(rule?.action).toBe('block');
    });

    it('blocks wrapped git reset --hard', () => {
      const { rule } = analyseCommand('bash -c "git reset --hard"');
      expect(rule?.action).toBe('block');
    });

    it('blocks nested wrappers', () => {
      const { rule } = analyseCommand('sudo env VAR=1 git reset --hard');
      expect(rule?.action).toBe('block');
    });
  });

  describe('warning operations', () => {
    it('warns on git reset --merge', () => {
      const { rule } = analyseCommand('git reset --merge');
      expect(rule?.action).toBe('warn');
      expect(rule?.severity).toBe('warning');
    });

    it('warns on git branch -D', () => {
      const { rule } = analyseCommand('git branch -D old-feature');
      expect(rule?.action).toBe('warn');
    });

    it('warns on git clean -f', () => {
      const { rule } = analyseCommand('git clean -f');
      expect(rule?.action).toBe('warn');
    });

    it('warns on git stash drop', () => {
      const { rule } = analyseCommand('git stash drop');
      expect(rule?.action).toBe('warn');
    });

    it('warns on git stash clear', () => {
      const { rule } = analyseCommand('git stash clear');
      expect(rule?.action).toBe('warn');
    });
  });

  describe('safe operations - ALLOWED', () => {
    it('allows git checkout -b new-branch', () => {
      const { rule } = analyseCommand('git checkout -b new-feature');
      expect(rule?.action).toBe('allow');
    });

    it('allows git checkout --orphan', () => {
      const { rule } = analyseCommand('git checkout --orphan orphan-branch');
      expect(rule?.action).toBe('allow');
    });

    it('allows git push --force-with-lease', () => {
      const { rule } = analyseCommand('git push --force-with-lease');
      expect(rule?.action).toBe('allow');
    });

    it('allows git branch -d (safe delete)', () => {
      const { rule } = analyseCommand('git branch -d merged-feature');
      expect(rule?.action).toBe('allow');
    });

    it('allows git restore --staged', () => {
      const { rule } = analyseCommand('git restore --staged file.txt');
      expect(rule?.action).toBe('allow');
    });

    it('allows git clean -n (dry-run)', () => {
      const { rule } = analyseCommand('git clean -n');
      expect(rule?.action).toBe('allow');
    });

    it('allows safe git operations', () => {
      const safeCommands = [
        'git status',
        'git log',
        'git diff',
        'git add .',
        'git commit -m "test"',
        'git pull',
        'git fetch',
      ];

      safeCommands.forEach((cmd) => {
        const { rule } = analyseCommand(cmd);
        // Should either be explicitly allowed or no rule matched (default allow)
        if (rule) {
          expect(rule.action).not.toBe('block');
        }
      });
    });
  });

  describe('rule coverage', () => {
    it('has at least 15 rules', () => {
      expect(DEFAULT_GIT_RULES.length).toBeGreaterThanOrEqual(15);
    });

    it('all rules have required fields', () => {
      DEFAULT_GIT_RULES.forEach((rule) => {
        expect(rule.id).toBeTruthy();
        expect(rule.command).toBe('git');
        expect(rule.action).toMatch(/^(block|warn|allow)$/);
        expect(rule.reason).toBeTruthy();
      });
    });

    it('block/warn rules have suggestions', () => {
      DEFAULT_GIT_RULES.filter((r) => r.action !== 'allow').forEach((rule) => {
        expect(rule.suggestion).toBeTruthy();
      });
    });
  });
});

import type { CommandRule } from './types.js';

export const DEFAULT_FILESYSTEM_RULES: CommandRule[] = [
  // === RM -RF BLOCKING ===

  {
    id: 'rm-rf-root',
    category: 'filesystem',
    command: 'rm',
    flags: { dangerous: ['-r', '-f', '--recursive', '--force'] },
    args: { pattern: /^(\/|~|~\/|\$HOME|\${HOME})$/ },
    action: 'block',
    severity: 'error',
    reason: 'rm -rf on root or home paths is extremely dangerous',
    suggestion:
      'NEVER delete root or home directories. Review the target path carefully.',
    references: [
      "https://en.wikipedia.org/wiki/Rm_(Unix)#Deletion_of_a_user's_home_directory",
    ],
  },

  {
    id: 'rm-rf-parent-traversal',
    category: 'filesystem',
    command: 'rm',
    flags: { dangerous: ['-r', '-f'] },
    args: { pattern: /\.\./ },
    action: 'block',
    severity: 'error',
    reason:
      'rm -rf with parent directory traversal (..) can escape current directory',
    suggestion: 'Use absolute paths or paths within the current directory only',
  },

  {
    id: 'rm-rf-current-dir',
    category: 'filesystem',
    command: 'rm',
    flags: { dangerous: ['-r', '-f'] },
    args: { pattern: /^\.$/ },
    action: 'block',
    severity: 'error',
    reason: 'rm -rf . deletes the entire current directory',
    suggestion: 'Specify exact subdirectories or files instead',
  },

  {
    id: 'rm-rf-general',
    category: 'filesystem',
    command: 'rm',
    flags: { dangerous: ['-r', '-f', '-rf', '-fr'] },
    action: 'block',
    severity: 'error',
    reason: 'rm -rf is destructive and can cause irreversible data loss',
    suggestion:
      'List files first with "ls -la", review carefully, then delete individually or use a safer method',

  // === SAFE EXCEPTIONS (higher specificity, checked first) ===

  {
    id: 'rm-rf-tmp-dir',
