## Objective

```typescript
export interface WatchState {
  status: 'idle' | 'running' | 'passing' | 'failing';
  currentRun: CurrentRunInfo | null;
  history: WatchRunResult[];
  statistics: WatchStatistics;
}

export interface CurrentRunInfo {
  files: string[];
  checksRunning: string[];
  checksCompleted: CheckResult[];
  startTime: Date;
}

export interface WatchRunResult {
  timestamp: Date;
  files: string[];
  duration: number;
  success: boolean;
  failedChecks?: string[];
}

export interface CheckResult {
  name: string;
  passed: boolean;
  duration: number;
}

export interface WatchStatistics {
  totalRuns: number;
  passed: number;
  failed: number;
  averageDuration: number;
}
```

### 2. Create Dashboard Component

**File:** `ui/src/components/WatchDashboard/WatchDashboard.tsx`

```typescript
import React, { useState, useEffect } from 'react';
import { Box, Text, useInput } from 'ink';
import { StatusBadge } from '../StatusBadge.js';
import { ProgressBar } from '../ProgressBar.js';
import { KeyboardShortcuts } from '../KeyboardShortcuts.js';
import type { WatchState, WatchRunResult, CheckResult } from './types.js';

export interface WatchDashboardProps {
  state: WatchState;
  onQuit: () => void;
  onRunNow?: () => void;
  onClearHistory?: () => void;
}

export const WatchDashboard: React.FC<WatchDashboardProps> = ({
  state,
  onQuit,
  onRunNow,
  onClearHistory,
}) => {
  useInput((input, key) => {
    if (input === 'q') {
      onQuit();
    } else if (input === 'r' && onRunNow) {
      onRunNow();
    } else if (input === 'c' && onClearHistory) {
      onClearHistory();
    }
  });

  return (
    <Box flexDirection="column" padding={1}>
      {/* Header */}
      <Box borderStyle="round" borderColor="cyan" padding={1} flexDirection="column">
        <Box justifyContent="space-between">
          <Text bold>Anvil Watch Mode</Text>
          <KeyboardShortcuts
            shortcuts={[
              { key: 'q', description: 'Quit' },
              { key: 'r', description: 'Run Now' },
              { key: 'c', description: 'Clear History' },
            ]}
          />
        </Box>
        <Box marginTop={1}>
          <Text>Status: </Text>
          <StatusBadge
            status={state.status === 'passing' ? 'passed' : state.status === 'failing' ? 'failed' : state.status}
          />
        </Box>
      </Box>

      {/* Current Run */}
      {state.currentRun && (
        <Box marginTop={1} borderStyle="round" borderColor="yellow" padding={1} flexDirection="column">
          <Text bold color="yellow">
            Current Run
          </Text>
          <Box marginTop={1}>
            <Text>
              Files: {state.currentRun.files.map((f) => f.split('/').pop()).join(', ')} (
              {state.currentRun.files.length} files)
            </Text>
          </Box>
          <Box marginTop={1} flexDirection="column">
            {state.currentRun.checksRunning.map((check) => (
              <ProgressBar key={check} label={check} progress={50} width={30} />
            ))}
          </Box>
          <Box marginTop={1}>
            <Text dimColor>
              Completed: {state.currentRun.checksCompleted.map((c) => c.name).join(', ')} (
              {state.currentRun.checksCompleted.length}/{state.currentRun.checksCompleted.length + state.currentRun.checksRunning.length})
            </Text>
          </Box>
        </Box>
      )}

      {/* History */}
      <Box marginTop={1} borderStyle="round" borderColor="blue" padding={1} flexDirection="column">
        <Text bold color="blue">
          History
        </Text>
        {state.history.length === 0 ? (
          <Text dimColor>No runs yet</Text>
        ) : (
          state.history.slice(0, 10).map((run, index) => (
            <Box key={index} marginTop={1}>
              <StatusBadge status={run.success ? 'passed' : 'failed'} />
              <Text> </Text>
              <Text dimColor>{run.timestamp.toLocaleTimeString()}</Text>
              <Text> </Text>
              <Text>{run.files.map((f) => f.split('/').pop()).join(', ')}</Text>
              <Text> </Text>
              <Text dimColor>{run.duration}ms</Text>
              {run.failedChecks && run.failedChecks.length > 0 && (
                <>
                  <Text> </Text>
                  <Text color="red">Failed: {run.failedChecks.join(', ')}</Text>
                </>
              )}
            </Box>
          ))
        )}
      </Box>

      {/* Statistics */}
      <Box marginTop={1} borderStyle="round" borderColor="green" padding={1} flexDirection="column">
        <Text bold color="green">
          Statistics
        </Text>
        <Box marginTop={1} justifyContent="space-between">
          <Text>
            Pass Rate: {state.statistics.totalRuns > 0 ? Math.round((state.statistics.passed / state.statistics.totalRuns) * 100) : 0}% ({state.statistics.passed}/
            {state.statistics.totalRuns})
          </Text>
          <Text>Avg Duration: {Math.round(state.statistics.averageDuration)}ms</Text>
        </Box>
      </Box>
    </Box>
  );
};
```

### 3. Create State Manager

**File:** `ui/src/components/WatchDashboard/WatchStateManager.ts`

```typescript
import type {
  WatchState,
  WatchRunResult,
  CurrentRunInfo,
  CheckResult,
} from './types.js';

export class WatchStateManager {
  private state: WatchState;
  private listeners: Set<(state: WatchState) => void>;

  constructor() {
    this.state = {
      status: 'idle',
      currentRun: null,
      history: [],
      statistics: {
        totalRuns: 0,
        passed: 0,
        failed: 0,
        averageDuration: 0,
      },
    };
    this.listeners = new Set();
  }

  subscribe(listener: (state: WatchState) => void): () => void {
    this.listeners.add(listener);
    return () => this.listeners.delete(listener);
  }

  getState(): WatchState {
    return { ...this.state };
  }

  startRun(files: string[]): void {
    this.state.status = 'running';
    this.state.currentRun = {
      files,
      checksRunning: [],
      checksCompleted: [],
      startTime: new Date(),
    };
    this.notify();
  }

  addCheckRunning(checkName: string): void {
    if (this.state.currentRun) {
      this.state.currentRun.checksRunning.push(checkName);
      this.notify();
    }
  }

  completeCheck(checkName: string, passed: boolean, duration: number): void {
    if (this.state.currentRun) {
      this.state.currentRun.checksRunning =
        this.state.currentRun.checksRunning.filter((c) => c !== checkName);
      this.state.currentRun.checksCompleted.push({
        name: checkName,
        passed,
        duration,
      });
      this.notify();
    }
  }

  completeRun(success: boolean, failedChecks?: string[]): void {
    if (!this.state.currentRun) return;

    const duration = Date.now() - this.state.currentRun.startTime.getTime();
    const result: WatchRunResult = {
      timestamp: this.state.currentRun.startTime,
      files: this.state.currentRun.files,
      duration,
      success,
      failedChecks,
    };

    this.state.history.unshift(result);
    if (this.state.history.length > 50) {
      this.state.history = this.state.history.slice(0, 50);
    }

    this.state.statistics.totalRuns++;
    if (success) {
      this.state.statistics.passed++;
    } else {
      this.state.statistics.failed++;
    }

    const totalDuration = this.state.history.reduce(
      (sum, r) => sum + r.duration,
      0
    );
    this.state.statistics.averageDuration =
      totalDuration / this.state.history.length;

    this.state.status = success ? 'passing' : 'failing';
    this.state.currentRun = null;
    this.notify();
  }

  clearHistory(): void {
    this.state.history = [];
    this.state.statistics = {
      totalRuns: 0,
      passed: 0,
      failed: 0,
      averageDuration: 0,
    };
    this.notify();
  }

  private notify(): void {
    this.listeners.forEach((listener) => listener(this.getState()));
  }
}
```

### 4. Integrate with Watch Command

**File:** `cli/src/commands/watch.ts` — Add TUI integration:

```typescript
import { shouldUseTUI } from '@anvil/ui';
import { render } from 'ink';
import React from 'react';
import {
  WatchDashboard,
  WatchStateManager,
} from '@anvil/ui/components/WatchDashboard';

// Inside action handler, before creating orchestrator:
const useTUI = shouldUseTUI({ tui: options.tui, noTui: options.noTui });

if (useTUI) {
  const stateManager = new WatchStateManager();

  // Render dashboard
  const { unmount } = render(
    React.createElement(WatchDashboard, {
      state: stateManager.getState(),
      onQuit: async () => {
        await orchestrator.stop();
        unmount();
        process.exit(0);
      },
      onRunNow: () => {
        // Trigger immediate run
      },
      onClearHistory: () => {
        stateManager.clearHistory();
      },
    })
  );

  // Subscribe to state changes
  stateManager.subscribe((state) => {
    // Re-render with new state
  });

  // Wire orchestrator events to state manager
  orchestrator.setValidateHandler(async (files: string[]) => {
    stateManager.startRun(files);
    stateManager.addCheckRunning('validate');

    const result = await originalValidateHandler(files);

    stateManager.completeCheck(
      'validate',
      result.success,
      result.executionTimeMs
    );
    stateManager.completeRun(
      result.success,
      result.success ? undefined : ['validate']
    );

    return result;
  });
}
```

### 6. Write Integration Tests

**File:** `ui/src/components/WatchDashboard/__tests__/WatchDashboard.test.tsx`

```typescript
import React from 'react';
import { render } from 'ink-testing-library';
import { describe, it, expect, vi } from 'vitest';
import { WatchDashboard } from '../WatchDashboard.js';
import type { WatchState } from '../types.js';

describe('WatchDashboard', () => {
  const mockState: WatchState = {
    status: 'idle',
    currentRun: null,
    history: [],
    statistics: {
      totalRuns: 0,
      passed: 0,
      failed: 0,
      averageDuration: 0,
    },
  };

  it('renders dashboard with idle status', () => {
    const { lastFrame } = render(
      <WatchDashboard state={mockState} onQuit={() => {}} />
    );
    expect(lastFrame()).toContain('Anvil Watch Mode');
    expect(lastFrame()).toContain('idle');
  });

  it('shows current run when active', () => {
    const activeState: WatchState = {
      ...mockState,
      status: 'running',
      currentRun: {
        files: ['src/test.ts'],
        checksRunning: ['lint', 'test'],
        checksCompleted: [],
        startTime: new Date(),
      },
    };
    const { lastFrame } = render(
      <WatchDashboard state={activeState} onQuit={() => {}} />
    );
    expect(lastFrame()).toContain('Current Run');
    expect(lastFrame()).toContain('lint');
    expect(lastFrame()).toContain('test');
  });

  it('displays history', () => {
    const historyState: WatchState = {
      ...mockState,
      history: [
        {
          timestamp: new Date(),
          files: ['src/test.ts'],
          duration: 250,
          success: true,
        },
      ],
    };
    const { lastFrame } = render(
      <WatchDashboard state={historyState} onQuit={() => {}} />
    );
    expect(lastFrame()).toContain('History');
    expect(lastFrame()).toContain('250ms');
  });

  it('calls onQuit when q pressed', () => {
    const onQuit = vi.fn();
    const { stdin } = render(
      <WatchDashboard state={mockState} onQuit={onQuit} />
    );
    stdin.write('q');
    expect(onQuit).toHaveBeenCalled();
  });
});
```


- CI environments
