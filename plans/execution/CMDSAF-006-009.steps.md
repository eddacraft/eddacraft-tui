// core/src/gate/checks/command-safety.check.ts

import type { GateCheck, CheckResult } from '../../types/gate.types.js';
import type { APSPlan } from '../../schema/aps.schema.js';
import { CommandParser } from '../parsers/command-parser.js';
import { RuleMatcher } from '../rules/rule-matcher.js';
import { DEFAULT_COMMAND_RULES } from '../rules/index.js';
import type {
  CommandSafetyConfig,
  CommandRule,
  CommandAnalysisResult,
} from '../rules/types.js';

export class CommandSafetyCheck implements GateCheck {
  name = 'command-safety';
  description = 'Validates shell commands for destructive operations';

  private parser: CommandParser;
  private matcher: RuleMatcher;

  constructor() {
    this.parser = new CommandParser();
    this.matcher = new RuleMatcher();
  }

  async run(plan: APSPlan, config?: CommandSafetyConfig): Promise<CheckResult> {
    // 1. Check if enabled
    if (config?.enabled === false) {
      return {
        check: this.name,
        status: 'passed',
        message: 'Command safety check disabled',
        details: {},
      };
    }

    // 2. Load rules (default + user config)
    const rules = this.loadRules(config);

    // 3. Extract commands from plan
    const commands = this.extractCommands(plan);

    if (commands.length === 0) {
      return {
        check: this.name,
        status: 'passed',
        message: 'No shell commands found in plan',
        details: { commandsAnalysed: 0 },
      };
    }

    // 4. Analyse each command
    const results: CommandAnalysisResult[] = [];
    for (const cmd of commands) {
      const parsed = this.parser.parse(cmd);
      const rule = this.matcher.findMatchingRule(parsed, rules);

      results.push({
        command: cmd,
        parsedCommand: parsed,
        matchedRule: rule,
        action: rule?.action || 'allow',
        severity: rule?.severity || 'info',
        reason: rule?.reason,
        suggestion: rule?.suggestion,
        references: rule?.references,
      });
    }

    // 5. Aggregate results
    const blocked = results.filter((r) => r.action === 'block');
    const warned = results.filter((r) => r.action === 'warn');

    return {
      check: this.name,
      status:
        blocked.length > 0
          ? 'failed'
          : warned.length > 0
            ? 'warning'
            : 'passed',
      message: this.formatSummary(
        blocked.length,
        warned.length,
        commands.length
      ),
      details: {
        blocked: blocked.map((r) => this.formatResult(r)),
        warnings: warned.map((r) => this.formatResult(r)),
        commandsAnalysed: commands.length,
      },
    };
  }

  private loadRules(config?: CommandSafetyConfig): CommandRule[] {
    let rules = [...DEFAULT_COMMAND_RULES];

    if (!config) {
      return rules;
    }

    // Apply disabled rules
    if (config.rules.disabled) {
      rules = rules.filter((r) => !config.rules.disabled!.includes(r.id));
    }

    // Apply overrides
    if (config.rules.overrides) {
      rules = rules.map((rule) => {
        const override = config.rules.overrides!.find((o) => o.id === rule.id);
        if (override) {
          return {
            ...rule,
            ...(override.action && { action: override.action }),
            ...(override.severity && { severity: override.severity }),
          };
        }
        return rule;
      });
    }

    // Add custom rules
    if (config.rules.custom) {
      rules.push(...config.rules.custom);
    }

    return rules;
  }

  private extractCommands(plan: APSPlan): string[] {
    const commands: string[] = [];

    // Extract from proposed_changes
    for (const change of plan.proposed_changes || []) {
      if ('command' in change && change.command) {
        commands.push(change.command as string);
      }
    }

    // Extract from metadata.executionSteps if present
    if (plan.metadata?.executionSteps) {
      for (const step of plan.metadata.executionSteps) {
        if (step.command) {
          commands.push(step.command);
        }
      }
    }

    return commands;
  }

  private formatSummary(
    blocked: number,
    warned: number,
    total: number
  ): string {
    if (blocked > 0) {
      return `Blocked ${blocked} dangerous command(s) out of ${total} analysed`;
    }
    if (warned > 0) {
      return `Found ${warned} potentially dangerous command(s) - review recommended`;
    }
    return `Analysed ${total} command(s), no dangerous operations detected`;
  }

  private formatResult(result: CommandAnalysisResult): any {
    return {
      command: result.command,
      reason: result.reason,
      suggestion: result.suggestion,
      references: result.references,
      severity: result.severity,
    };
  }
}

// core/src/gate/checks/command-safety.check.test.ts

import { describe, it, expect } from 'vitest';
import { CommandSafetyCheck } from './command-safety.check.js';
import type { APSPlan } from '../../schema/aps.schema.js';

describe('CommandSafetyCheck', () => {
  const check = new CommandSafetyCheck();

  function createPlanWithCommands(commands: string[]): APSPlan {
    return {
      schema_version: '0.1.0',
      id: 'test-plan',
      hash: 'test-hash',
      intent: 'test',
      proposed_changes: commands.map((cmd) => ({
        type: 'command',
        command: cmd,
      })),
      provenance: {
        created_at: new Date().toISOString(),
        created_by: 'test',
      },
      context: {},
      validation: {},
      evidence: [],
    } as APSPlan;
  }

  describe('basic functionality', () => {
    it('passes when no commands present', async () => {
      const plan = createPlanWithCommands([]);
      const result = await check.run(plan);

      expect(result.status).toBe('passed');
      expect(result.message).toContain('No shell commands');
    });

    it('passes with safe commands', async () => {
      const plan = createPlanWithCommands([
        'git status',
        'npm install',
        'pnpm build',
      ]);
      const result = await check.run(plan);

      expect(result.status).toBe('passed');
    });
  });

  describe('blocking dangerous commands', () => {
    it('blocks git reset --hard', async () => {
      const plan = createPlanWithCommands(['git reset --hard']);
      const result = await check.run(plan);

      expect(result.status).toBe('failed');
      expect(result.details.blocked).toHaveLength(1);
      expect(result.details.blocked[0].reason).toContain(
        'destroys uncommitted changes'
      );
    });

    it('blocks rm -rf /', async () => {
      const plan = createPlanWithCommands(['rm -rf /']);
      const result = await check.run(plan);

      expect(result.status).toBe('failed');
      expect(result.details.blocked[0].reason).toContain('extremely dangerous');
    });

    it('blocks multiple dangerous commands', async () => {
      const plan = createPlanWithCommands([
        'git reset --hard',
        'git push --force',
        'rm -rf ~',
      ]);
      const result = await check.run(plan);

      expect(result.status).toBe('failed');
      expect(result.details.blocked).toHaveLength(3);
    });
  });

  describe('configuration', () => {
    it('respects disabled rules', async () => {
      const plan = createPlanWithCommands(['git reset --hard']);
      const result = await check.run(plan, {
        enabled: true,
        strict: false,
        rules: {
          disabled: ['git-reset-hard'],
        },
      });

      expect(result.status).toBe('passed');
    });

    it('applies overrides', async () => {
      const plan = createPlanWithCommands(['git push --force']);
      const result = await check.run(plan, {
        enabled: true,
        strict: false,
        rules: {
          overrides: [{ id: 'git-push-force', action: 'warn' }],
        },
      });

      expect(result.status).toBe('warning');
      expect(result.details.warnings).toHaveLength(1);
    });

    it('can be disabled via config', async () => {
      const plan = createPlanWithCommands(['git reset --hard']);
      const result = await check.run(plan, {
        enabled: false,
        strict: false,
        rules: {},
      });

      expect(result.status).toBe('passed');
      expect(result.message).toContain('disabled');
    });
  });
});

// core/src/gate/config/command-safety-config.ts

import type { CommandSafetyConfig } from '../rules/types.js';

export const DEFAULT_COMMAND_SAFETY_CONFIG: CommandSafetyConfig = {
  enabled: true,
  strict: false,
  rules: {
    overrides: [],
    custom: [],
    disabled: [],
  },
  workingDirectory: {
    allowDeleteInCwd: false,
    tempDirPatterns: ['/tmp', '/var/tmp', '$TMPDIR'],
  },
  output: {
    verbose: false,
    showSuggestions: true,
    showReferences: true,
  },
};

export function loadCommandSafetyConfig(
  userConfig?: Partial<CommandSafetyConfig>
): CommandSafetyConfig {
  if (!userConfig) {
    return DEFAULT_COMMAND_SAFETY_CONFIG;
  }

  return {
    enabled: userConfig.enabled ?? DEFAULT_COMMAND_SAFETY_CONFIG.enabled,
    strict: userConfig.strict ?? DEFAULT_COMMAND_SAFETY_CONFIG.strict,
    rules: {
      overrides: userConfig.rules?.overrides || [],
      custom: userConfig.rules?.custom || [],
      disabled: userConfig.rules?.disabled || [],
    },
    workingDirectory: {
      allowDeleteInCwd:
        userConfig.workingDirectory?.allowDeleteInCwd ??
        DEFAULT_COMMAND_SAFETY_CONFIG.workingDirectory!.allowDeleteInCwd,
      tempDirPatterns:
        userConfig.workingDirectory?.tempDirPatterns ||
        DEFAULT_COMMAND_SAFETY_CONFIG.workingDirectory!.tempDirPatterns,
    },
    output: {
      verbose:
        userConfig.output?.verbose ??
        DEFAULT_COMMAND_SAFETY_CONFIG.output!.verbose,
      showSuggestions:
        userConfig.output?.showSuggestions ??
        DEFAULT_COMMAND_SAFETY_CONFIG.output!.showSuggestions,
      showReferences:
        userConfig.output?.showReferences ??
        DEFAULT_COMMAND_SAFETY_CONFIG.output!.showReferences,
    },
  };
}

// core/src/gate/formatters/command-safety-formatter.ts

import type { CommandAnalysisResult } from '../rules/types.js';

export class CommandSafetyFormatter {
  formatBlockedCommand(result: CommandAnalysisResult): string {
    const lines = [`❌ ${result.command}`, `   Reason: ${result.reason}`];

    if (result.suggestion) {
      lines.push(`   Suggestion: ${result.suggestion}`);
    }

    if (result.references && result.references.length > 0) {
      lines.push(`   Reference: ${result.references[0]}`);
    }

    return lines.join('\n');
  }

  formatWarning(result: CommandAnalysisResult): string {
    const lines = [`⚠️  ${result.command}`, `   Reason: ${result.reason}`];

    if (result.suggestion) {
      lines.push(`   Suggestion: ${result.suggestion}`);
    }

    return lines.join('\n');
  }

  formatSummary(blocked: number, warned: number, total: number): string {
    if (blocked > 0) {
      return `Command Safety Check Failed: ${blocked} dangerous command(s) blocked`;
    }
    if (warned > 0) {
      return `Command Safety Warnings: ${warned} potentially dangerous command(s)`;
    }
    return `Command Safety: All ${total} commands safe`;
  }
}

# Command Safety Validation

Anvil's command safety system prevents data loss from destructive shell
commands.

## Quick Start

Command safety runs automatically with `anvil gate`:

\`\`\`bash anvil gate plan.md \`\`\`

## What It Blocks

### Git Operations

- `git reset --hard` — Destroys uncommitted changes
- `git push --force` — Rewrites remote history
- `git checkout --` — Discards working tree changes

### Filesystem Operations

- `rm -rf /` — Deletes root
- `rm -rf ~` — Deletes home directory

## What It Allows

- `git checkout -b` — Branch creation
- `git push --force-with-lease` — Safer force push
- `rm -rf node_modules` — Reproducible artifacts
